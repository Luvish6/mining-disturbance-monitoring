<script>
const layers = __LAYERS__;
const select = document.getElementById("layer");
const map = document.getElementById("map");
const zoom = document.getElementById("zoom");
for (const [index, layer] of layers.entries()) select.add(new Option(layer.title, index));
function render() {
  const layer = layers[Number(select.value)];
  map.src = layer.key + ".png";
  map.alt = layer.title;
  document.getElementById("period").textContent = layer.period;
  document.getElementById("legend").textContent = layer.legend;
  const swatch = document.getElementById("swatch");
  swatch.hidden = !layer.colors.length;
  swatch.style.background = layer.colors.length ? "linear-gradient(90deg," + layer.colors.join(",") + ")" : "";
}
select.addEventListener("change", render);
zoom.addEventListener("click", () => {
  const pressed = zoom.getAttribute("aria-pressed") === "true";
  zoom.setAttribute("aria-pressed", String(!pressed));
  zoom.textContent = pressed ? "Magnify" : "Fit width";
  map.style.width = pressed ? "100%" : "200%";
});
render();
</script>
